! =========================================================
! LCW PROJESİ - ROUTER0 YAPILANDIRMA DOSYASI
! Amaç: VLAN'lar arası yönlendirme, Router-on-a-Stick ve ACL
! =========================================================

hostname Router

ip cef
no ipv6 cef

! Ana fiziksel arayüz IP almaz, alt arayüzler VLAN gateway olarak çalışır.
interface GigabitEthernet0/0
 no ip address
 duplex auto
 speed auto
 no shutdown

! VLAN 10 - Yönetim
interface GigabitEthernet0/0.10
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0

! VLAN 20 - İnsan Kaynakları
interface GigabitEthernet0/0.20
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0

! VLAN 30 - Muhasebe
interface GigabitEthernet0/0.30
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
 ip access-group 101 in

! VLAN 40 - Satış
interface GigabitEthernet0/0.40
 encapsulation dot1Q 40
 ip address 192.168.40.1 255.255.255.0

! VLAN 50 - E-Ticaret
interface GigabitEthernet0/0.50
 encapsulation dot1Q 50
 ip address 192.168.50.1 255.255.255.0

! VLAN 60 - Lojistik
interface GigabitEthernet0/0.60
 encapsulation dot1Q 60
 ip address 192.168.60.1 255.255.255.0

! VLAN 70 - IT
interface GigabitEthernet0/0.70
 encapsulation dot1Q 70
 ip address 192.168.70.1 255.255.255.0

! VLAN 80 - Sunucular
interface GigabitEthernet0/0.80
 encapsulation dot1Q 80
 ip address 192.168.80.1 255.255.255.0

! VLAN 90 - DMZ
interface GigabitEthernet0/0.90
 encapsulation dot1Q 90
 ip address 192.168.90.1 255.255.255.0

interface GigabitEthernet0/1
 no ip address
 shutdown

! ACL 101:
! Muhasebe VLAN'ının DMZ ağına erişimi engellenir.
! Diğer tüm trafik izinlidir.
access-list 101 deny ip 192.168.30.0 0.0.0.255 192.168.90.0 0.0.0.255
access-list 101 permit ip any any

end
write memory
